%  Steady state Identification of non Linear model::::

function y = CSTR_NonLinear(t, CA, F, CA0, V, K)

CA1 = CA(1);
CA2 = CA(2);
y(1) = (F/V)*(CA0 - CA1) - K*V*CA1;
y(2) = (F/V)*(CA1 - CA2) - K*V*CA2;
y = y';
end

%--------------------------now for Tau time constant how to compute------
function tau_val = find_tau(time_vec, y_vec, t_start, t_end)
    y_start  = y_vec(t_start);
    y_final  = y_vec(t_end);
    y_target = y_start + 0.632 * (y_final - y_start);   % 63.2% level
    idx_tau  = find(y_vec >= y_target, 1, 'first');     % find first crossing point
    if isempty(idx_tau)
        tau_val = NaN;
    else
        tau_val = time_vec(idx_tau) - time_vec(t_start); % elapsed time since step start
    end
end

%---------------------LINEARISATION USING JACOBIAN METHOD--------------

% making function for jacobian 

function y = CSTR_Jacobian(X) 
    global V K 
    CA1 = X(1,1);
    CA2 = X(2,1);
    CA0 = X(3,1);
    F   = X(4,1);
    y(1) = ((F/V)*(CA0 - CA1)) - (K*V*CA1);
    y(2) = ((F/V)*(CA1 - CA2)) - (K*V*CA2);
    y = y';
end
%----linearisation------
X_vec = [CA1_init_guess; CA2_init_guess; CA0 ; F]; % Linearizing w.r.t steady states of state variable (CA1 ,CA2), Manipulated input (CAO) and Disturbance Variable (F)
  
%-------------JACOBIAN MATRIX--------------
function gradF = Jacobian(myFunc, X_vec)
    l = length(X_vec);          % number of variables
    for i = 1:1:l
        delta = X_vec(i) / 1e+4; % small perturbation
        if delta == 0
            delta = 1e-4;
        end
    
        X_plus = X_vec; 
        X_plus(i) = X_plus(i) + delta;
        X_minus = X_vec;
        X_minus(i) = X_minus(i) - delta;
    
        f_plus = feval(myFunc, X_plus);
        f_minus = feval(myFunc, X_minus);
    
        gradF(:,i) = (f_plus - f_minus) / (2*delta);
    end
end

J = Jacobian( @CSTR_Jacobian, X_vec)


function y = CSTR_Linear(t,CA_dev,CA0_dev,F_dev,J)
    CA1_dev = CA_dev(1);
    CA2_dev = CA_dev(2);
    y(1) = (J(1,1)*CA1_dev) + (J(1,2)*CA2_dev) + (J(1,3)*CA0_dev) + (J(1,4)*F_dev);
    y(2) = (J(2,1)*CA1_dev) + (J(2,2)*CA2_dev) + (J(2,3)*CA0_dev) + (J(2,4)*F_dev);
    
    y = y';
end

% Build state-space matrices from Jacobian J
A_mat = J(:,1:2);    % 2x2: derivatives w.r.t CA1, CA2

%  output matrix for CA1 and CA2

Ca2 = [0 1];      % output C_A2

% construct space state system for CA2
