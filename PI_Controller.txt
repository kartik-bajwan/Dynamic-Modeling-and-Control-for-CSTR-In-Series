clear all
clc

%% Step 1: setting parameter given in the problem 
global CA10 CA20 CA0i F V k

CA10 = 0.236;   % initial guess for CA1
CA20 = 0.05;    % initial guess for CA2
CA0i = 0.925;   % manipulated variable (CA0)
F    = 0.085;   % disturbance variable
V    = 2.1;     % reactor volume
k    = 0.040;   % rate constant

%% -------- Steady State Identification

X0 = [CA10; CA20];       % initial state [CA1; CA2]

load("steadyState.mat")  %load steady state data from main file (Assignment 1)

CA_value(1) = CA20;      
time(1)     = 0;
y0          = X0;      

%% steady state setpoint

y_setPoint = CA_ss(1,2);       % CA2 steady-state

%% Load PI controller parameters (CA2 output, CA0 input, F disturbance)

load("parameters.mat");  % extract Kp, TauP, Theta from here 

Tau_c = TauP / 2;

Kc   = (1 / Kp) * (TauP / (Tau_c + Theta));   % controller gain
TauI = TauP;                                  % integral time

CA0_value(1)   = CA0i;    % guess initial given for manipulated varaible 
time_series(1) = 0;       


%   PI controller
% for 3000 samples 

for i = 1:1:3000
   
    % Setpoint change (servo problem)
    if i <= 500
        Y_setPoint(i) = y_setPoint;
    else
        Y_setPoint(i) = 1.05 * y_setPoint;
    end

    % Disturbance change (regulatory problem)
    if i >= 2000
        distVar(i) = 1.05 * F;
    else
        distVar(i) = F;
    end
   
    % Nonlinear CSTR model for 1 min sampling time it takes
    [t, Y] = ode45(@(t, CA) CSTR_Non_Linear(t, CA, CA0_value(i), distVar(i)), ...
                   [0, 1], y0);
    y0 = Y(end, :).';      % update state for next iteration

    % Store CA2 and time
    CA_value(i+1) = y0(2);
    time(i+1)     = i;

    % PI controller error (same structure as your corrected code)
    error(i) = Y_setPoint(i) - y0(2);

    delta = error(i) + (1/TauI) * sum(error(1:i));

    % Update manipulated variable
    CA0_value(i+1) = CA0i + Kc * delta;

end


%% Plotting

figure(1)
sgtitle('Feedback control design for PI controller'); 
subplot(2,2,1)

xlabel('Time (s)');

grid on;

subplot(2,2,2)
plot(time(2:end), CA0_value(2:end), 'b-', 'LineWidth', 1.5);
xlabel('Time (s)');
ylabel('CA0');
title('CA0 vs Time');
grid on;

subplot(2,2,3)

xlabel('Time (s)');
ylabel('Setpoint');
title('Setpoint vs Time');
grid on;

subplot(2,2,4)
plot(time(2:end), distVar, 'm-', 'LineWidth', 1.5);
xlabel('Time (s)');
ylabel('Disturbance Variable');
title('Disturbance Variable vs Time');
grid on;


%% Nonlinear CSTR model

function y = CSTR_Non_Linear(t, CA, CA0, F)

    global V k

    CA1 = CA(1);
    CA2 = CA(2);

    y1 = ((F / V) * (CA0 - CA1)) - (V * k * CA1);
    y2 = ((F / V) * (CA1 - CA2)) - (V * k * CA2);

    y = [y1; y2];

end
